🎯 เมื่อเรียนจบบทนี้ นิสิตจะสามารถ
  1. เขียน INSERT / UPDATE / DELETE ที่ใช้ผลลัพธ์จาก SELECT และ CASE ได้
  2. สร้างและใช้งานวิว และระบุได้ว่าวิวใดปรับปรุงข้อมูลผ่านได้
  3. อธิบายหลักการของดัชนี และเลือกสร้างดัชนีได้เหมาะสม
  4. อธิบาย ACID และควบคุมทรานแซกชันด้วย COMMIT, ROLLBACK, SAVEPOINT ได้
  5. อธิบายปัญหาการทำงานพร้อมกันและระดับการแยกรายการได้

7.1 การจัดการข้อมูลด้วยผลลัพธ์จากคำสั่งสอบถาม

7.1.1 INSERT … SELECT

คัดลอกข้อมูลจากตารางหนึ่งไปอีกตารางโดยไม่ต้องพิมพ์ทีละแถว — จำนวนคอลัมน์และชนิดข้อมูลต้องสอดคล้องกัน

CREATE TABLE student_honor ( stdid VARCHAR(10) PRIMARY KEY, name VARCHAR(50), major VARCHAR(20), gpa DECIMAL(3,2), moved_date DATE ); INSERT INTO student_honor (stdid, name, major, gpa, moved_date) SELECT stdid, name, major, gpa, CURDATE() FROM student WHERE gpa >= 3.50;

CREATE TABLE … AS SELECT สร้างตาราง + คัดลอกข้อมูลในคำสั่งเดียว แต่ไม่คัดลอกคีย์หลักและข้อกำหนดบังคับ

CREATE TABLE enroll_backup AS SELECT * FROM enroll; DESCRIBE enroll_backup;

ต่างจาก enroll ต้นฉบับที่ secid, stdid เป็น PRI — FK และ CHECK ของเกรดก็ไม่ถูกคัดลอก ถ้าจะใช้เป็นตารางถาวรต้องเพิ่มเอง

ALTER TABLE enroll_backup ADD PRIMARY KEY (secid, stdid);

จงสร้างตาราง top_student(stdid, name, gpa) แล้วคัดลอกนิสิตที่มี gpa ตั้งแต่ 3.00 ขึ้นไปเข้าไป

CREATE TABLE top_student ( stdid VARCHAR(10) PRIMARY KEY, name VARCHAR(50), gpa DECIMAL(3,2) ); INSERT INTO top_student SELECT stdid, name, gpa FROM student WHERE gpa >= 3.00;

7.1.2 UPDATE โดยอ้างอิงตารางอื่น

ลด gpa ลง 5% ให้นิสิตที่เคยได้ F ตั้งแต่ 2 ครั้ง — ตรวจกลุ่มเป้าหมายก่อนเสมอ

SELECT stdid, COUNT(*) AS n_fail FROM enroll WHERE grade = 'F' GROUP BY stdid HAVING COUNT(*) >= 2; UPDATE student SET gpa = gpa * 0.95 WHERE stdid IN (SELECT stdid FROM enroll WHERE grade = 'F' GROUP BY stdid HAVING COUNT(*) >= 2);

UPDATE … JOIN (ไวยากรณ์เฉพาะ MySQL) — ขึ้นเงินเดือนอาจารย์กลุ่มละ 500 บาทตามจำนวนกลุ่มที่สอน

UPDATE lecturer l INNER JOIN (SELECT lecid, COUNT(*) AS n FROM section GROUP BY lecid) t ON l.lecid = t.lecid SET l.salary = l.salary + (t.n * 500) WHERE l.salary IS NOT NULL;

7.1.3 DELETE โดยอ้างอิงตารางอื่น

DELETE FROM enroll WHERE secid IN (SELECT secid FROM section WHERE term = '2023-1');
⚠️ ข้อจำกัดของ MySQL (Error 1093)

Subquery อ้างอิงตารางเดียวกับที่กำลังแก้/ลบไม่ได้ — แก้โดยครอบ Subquery อีกชั้นให้เป็น Derived Table

DELETE FROM student WHERE gpa < (SELECT AVG(gpa) FROM student); DELETE FROM student WHERE gpa < (SELECT avg_gpa FROM (SELECT AVG(gpa) AS avg_gpa FROM student) AS t);

จงเพิ่มเงินเดือน 3,000 บาทให้อาจารย์ที่สอนมากกว่า 5 กลุ่มเรียน

UPDATE lecturer SET salary = salary + 3000 WHERE lecid IN (SELECT lecid FROM section GROUP BY lecid HAVING COUNT(*) > 5);

ตรวจด้วย SELECT lecid, COUNT(*) FROM section GROUP BY lecid ก่อน จะเห็นว่ามี t01 (14 กลุ่ม), t02 (11), t03 (8) และ t05 (6) ที่สอนเกิน 5 กลุ่ม — t04 สอน 4 กลุ่ม แสดงไว้เทียบว่าไม่เปลี่ยน

จงลบการลงทะเบียนที่ยังไม่มีเกรด ในกลุ่มเรียนของภาคเรียน 2025-1

DELETE FROM enroll WHERE grade IS NULL AND secid IN (SELECT secid FROM section WHERE term = '2025-1');

กลุ่มเรียนภาค 2025-1 คือ secid 34–38 — มีแถวที่ grade เป็น NULL ใน 34, 35, 36, 37

7.2 การแก้ไขข้อมูลตามเงื่อนไขด้วย CASE

CASE ใช้ใน SET ของ UPDATE ได้ แก้หลายเงื่อนไขในคำสั่งเดียว — เงื่อนไขตรวจจากบนลงล่าง และหยุดที่ข้อแรกที่เป็นจริง

SELECT lecid, name, salary FROM lecturer WHERE salary IS NOT NULL ORDER BY lecid; UPDATE lecturer SET salary = CASE WHEN salary < 30000 THEN salary * 1.15 WHEN salary BETWEEN 30000 AND 60000 THEN salary * 1.10 ELSE salary * 1.05 END WHERE salary IS NOT NULL;

จงเพิ่มคอลัมน์ status ในตาราง student แล้วกำหนดค่า “พ้นสภาพ” เมื่อ gpa < 1.60, “รอพินิจ” เมื่อ 1.60–1.99, “ปกติ” กรณีอื่น (gpa เป็น NULL ให้เป็น “ยังไม่มีผล”)

ALTER TABLE student ADD COLUMN status VARCHAR(20); UPDATE student SET status = CASE WHEN gpa IS NULL THEN 'ยังไม่มีผล' WHEN gpa < 1.60 THEN 'พ้นสภาพ' WHEN gpa < 2.00 THEN 'รอพินิจ' ELSE 'ปกติ' END; ALTER TABLE student DROP COLUMN status;

7.3 วิว (View)

วิว คือตารางเสมือนจาก SELECT — เก็บเฉพาะคำสั่ง ไม่เก็บข้อมูล เรียกใช้เมื่อไรก็ประมวลผลกับตารางจริงขณะนั้น จึงได้ข้อมูลล่าสุดเสมอ

แนวคิดการทำงานของวิว
  1. ซ่อนความซับซ้อน — JOIN 4–5 ตาราง เก็บเป็นวิวแล้ว SELECT ธรรมดา
  2. ควบคุมความปลอดภัย — ให้สิทธิ์เฉพาะวิว ไม่ให้แตะตารางจริง (บทที่ 8)
  3. มุมมองเฉพาะกลุ่ม — คือ External Level ในสถาปัตยกรรม 3 ระดับ (บทที่ 1)
  4. ลดการเขียนซ้ำ — แก้ตรรกะที่วิวจุดเดียว ทุกโปรแกรมได้ผลทันที
CREATE VIEW v_cs_student AS SELECT stdid, name, gpa, birthday FROM student WHERE major = 'CS'; SELECT * FROM v_cs_student WHERE gpa >= 2.50; CREATE VIEW v_enroll_detail AS SELECT s.stdid AS student_id, s.name AS student_name, sub.subid AS subject_id, sub.name AS subject_name, sub.credit, l.name AS lecturer_name, sec.term, e.grade FROM enroll e INNER JOIN student s ON e.stdid = s.stdid INNER JOIN section sec ON e.secid = sec.secid INNER JOIN subject sub ON sec.subid = sub.subid INNER JOIN lecturer l ON sec.lecid = l.lecid; SELECT * FROM v_enroll_detail ORDER BY student_id, term; SELECT student_id, student_name, SUM(credit) AS total_credit, ROUND(SUM(credit * CASE grade WHEN 'A' THEN 4 WHEN 'B' THEN 3 WHEN 'C' THEN 2 WHEN 'D' THEN 1 ELSE 0 END) / SUM(credit), 2) AS gpa_calc FROM v_enroll_detail WHERE grade IS NOT NULL GROUP BY student_id, student_name ORDER BY gpa_calc DESC;
📝 ข้อสังเกต

gpa_calc ไม่ตรงกับคอลัมน์ student.gpa — เพราะ gpa ในตารางเป็นข้อมูลที่กรอกไว้ (ดีนอร์มัลไลซ์) ไม่ได้คำนวณจากการลงทะเบียนจริง นี่คือความเสี่ยงของการเก็บค่าที่คำนวณได้ (บทที่ 2–3)

7.3.1 การปรับปรุงข้อมูลผ่านวิว

ลักษณะของวิวปรับปรุงผ่านได้?เหตุผล
ตารางเดียว เลือกบางคอลัมน์/บางแถว✔ ได้ระบุแถวในตารางจริงได้ชัดเจน
มีฟังก์ชันรวม / GROUP BY / HAVING✘ ไม่ได้1 แถวในวิวมาจากหลายแถว
มี DISTINCT✘ ไม่ได้ไม่รู้ว่าตรงกับแถวใด
JOIN หลายตารางบางกรณีแก้ได้เฉพาะคอลัมน์จากตารางเดียว
มี UNION✘ ไม่ได้ไม่รู้ว่ามาจากชุดใด
UPDATE v_cs_student SET gpa = 3.60 WHERE stdid = '60003';

WITH CHECK OPTION ป้องกันการแก้ข้อมูลจนแถวหลุดออกจากเงื่อนไขของวิว

CREATE VIEW v_cs_student2 AS SELECT stdid, name, major, gpa FROM student WHERE major = 'CS' WITH CHECK OPTION; UPDATE v_cs_student2 SET major = 'Math' WHERE stdid = '60003'; DROP VIEW IF EXISTS v_cs_student2;

จงสร้างวิว v_lecturer_load แสดงรหัส ชื่ออาจารย์ จำนวนกลุ่มเรียนที่สอน และจำนวนนิสิตรวม แล้วแสดงอาจารย์ที่มีนิสิตรวมสูงสุด 3 อันดับ — วิวนี้ UPDATE ผ่านได้หรือไม่

CREATE VIEW v_lecturer_load AS SELECT l.lecid, l.name, COUNT(DISTINCT sec.secid) AS n_section, COUNT(e.stdid) AS n_student FROM lecturer l LEFT JOIN section sec ON l.lecid = sec.lecid LEFT JOIN enroll e ON sec.secid = e.secid GROUP BY l.lecid, l.name; SELECT * FROM v_lecturer_load ORDER BY n_student DESC LIMIT 3;

UPDATE ผ่านไม่ได้ เพราะมี GROUP BY และฟังก์ชันรวม — ใช้ COUNT(DISTINCT sec.secid) เพราะหลัง JOIN กับ enroll กลุ่มเรียนหนึ่งจะซ้ำหลายแถว

จงสร้างวิวที่แสดงเฉพาะรหัส ชื่อ สาขาของนิสิต (ไม่แสดง gpa และวันเกิด) และอธิบายประโยชน์ด้านความปลอดภัย

CREATE VIEW v_student_public AS SELECT stdid, name, major FROM student; SELECT * FROM v_student_public LIMIT 5;

ให้สิทธิ์ SELECT เฉพาะวิวนี้แก่ผู้ใช้ทั่วไป โดยไม่ให้สิทธิ์ตาราง student — ผู้ใช้จะไม่เห็น gpa และ birthday เลย (บทที่ 8)

7.4 ดัชนี (Index) และการเพิ่มประสิทธิภาพ

ดัชนี คือโครงสร้างเสริม (MySQL ใช้ B-Tree) ช่วยค้นหาเร็วขึ้น เหมือนดัชนีท้ายเล่มหนังสือ — ไม่มีดัชนีต้อง Full Table Scan (อ่านทุกแถว)

EXPLAIN SELECT stdid, name FROM student WHERE major = 'CS'; CREATE INDEX idx_student_major ON student(major); EXPLAIN SELECT stdid, name FROM student WHERE major = 'CS'; -- ดัชนีหลายคอลัมน์ (ลำดับคอลัมน์มีผล) CREATE INDEX idx_section_term_lec ON section(term, lecid); -- ดัชนีที่บังคับไม่ให้ค่าซ้ำ CREATE UNIQUE INDEX uidx_subject_name ON subject(name); SHOW INDEX FROM student; DROP INDEX idx_student_major ON student;

ควรสร้างดัชนีเมื่อ

  • ใช้ใน WHERE บ่อย
  • ใช้เป็นเงื่อนไข JOIN
  • ใช้ใน ORDER BY / GROUP BY
  • ต้องบังคับไม่ให้ซ้ำ

ไม่ควรสร้างเมื่อ

  • ตารางเล็กมาก
  • ค่าซ้ำกันมาก เช่น เพศ
  • ตารางถูกเพิ่ม/แก้ถี่มาก
  • แทบไม่เคยใช้ในเงื่อนไข
📝 ต้นทุนของดัชนี

อ่านเร็วขึ้น แต่ INSERT/UPDATE/DELETE ช้าลง (ต้องปรับดัชนีด้วย) และใช้เนื้อที่เพิ่ม — PK และ UNIQUE มีดัชนีให้อัตโนมัติอยู่แล้ว (สังเกต PRIMARY ในผล SHOW INDEX)

ในฐานข้อมูลกรณีศึกษา ควรสร้างดัชนีให้คอลัมน์ใด และไม่ควรสร้างที่ใด พร้อมเหตุผล

  • section.term — ค้นตามภาคเรียนบ่อย
  • enroll.stdid — ใช้ JOIN/ค้นทรานสคริปต์ (FK ใน MySQL มีดัชนีให้อัตโนมัติอยู่แล้ว)
  • student.name — ถ้ามีการค้นด้วยชื่อ
  • ไม่ควร: enroll.grade (มีแค่ 5 ค่า ซ้ำกันมาก), subject.credit (ค่าน้อยและไม่ค่อยใช้ค้น)

7.5 ทรานแซกชันและคุณสมบัติ ACID

ทรานแซกชัน คือกลุ่มคำสั่งที่ต้องทำงานเป็นหน่วยเดียว — สำเร็จทั้งหมด หรือไม่สำเร็จเลย เช่น โอนเงิน ต้องทั้งหักต้นทางและเพิ่มปลายทาง

สถานะการทำงานของทรานแซกชัน
คุณสมบัติความหมายตัวอย่างในระบบลงทะเบียน
Atomicityสำเร็จทั้งหมดหรือยกเลิกทั้งหมดเปิด section ใหม่ + ลงทะเบียนนิสิต ต้องสำเร็จพร้อมกัน
Consistencyฐานข้อมูลถูกต้องตามกฎทั้งก่อนและหลังจำนวนที่นั่งคงเหลือต้องไม่ติดลบ
Isolationทรานแซกชันพร้อมกันไม่รบกวนกัน2 คนแย่งที่นั่งสุดท้าย ต้องได้คนเดียว
Durabilityยืนยันแล้วข้อมูลต้องคงอยู่แม้ระบบล่มแจ้งว่าลงสำเร็จแล้ว ไฟดับข้อมูลต้องไม่หาย

7.5.1 คำสั่งควบคุมทรานแซกชัน

MySQL ทำงานแบบ autocommit (ทุกคำสั่งยืนยันทันที) — ถ้าจะควบคุมเองต้องเริ่มด้วย START TRANSACTION แล้วจบด้วย COMMIT หรือ ROLLBACK

START TRANSACTION; DELETE FROM enroll; SELECT COUNT(*) AS rows_in_enroll FROM enroll; ROLLBACK; SELECT COUNT(*) AS rows_in_enroll FROM enroll; START TRANSACTION; INSERT INTO section (subid, lecid, term) VALUES ('CS010', 't14', '2026-1'); INSERT INTO enroll (secid, stdid, grade) VALUES (LAST_INSERT_ID(), '60004', NULL); SELECT * FROM section WHERE term = '2026-1'; COMMIT;

SAVEPOINT ย้อนกลับเพียงบางส่วน

START TRANSACTION; UPDATE lecturer SET salary = salary + 1000 WHERE major = 'CS'; SAVEPOINT after_cs; UPDATE lecturer SET salary = salary + 5000 WHERE major = 'DC'; ROLLBACK TO SAVEPOINT after_cs; -- ยกเลิกเฉพาะของสาขา DC COMMIT; -- ยืนยันเฉพาะของสาขา CS SELECT lecid, name, salary, major FROM lecturer WHERE major IN ('CS', 'DC');
⚠️ Implicit Commit

คำสั่ง DDL (CREATE / ALTER / DROP TABLE) ทำให้ MySQL COMMIT ทรานแซกชันที่ค้างอยู่อัตโนมัติ และย้อนกลับไม่ได้ — แยก DDL ออกจากทรานแซกชันเสมอ

จงเขียนทรานแซกชันที่ (1) เปิดกลุ่มเรียนใหม่วิชา MA003 อาจารย์ t15 ภาค 2026-1 (2) ลงทะเบียนนิสิต 60014 และ 60015 ในกลุ่มนั้น และถ้าขั้นตอนใดล้มเหลวให้ยกเลิกทั้งหมด

START TRANSACTION; INSERT INTO section (subid, lecid, term) VALUES ('MA003', 't15', '2026-1'); SET @new_sec = LAST_INSERT_ID(); INSERT INTO enroll (secid, stdid) VALUES (@new_sec, '60014'), (@new_sec, '60015'); COMMIT; SELECT * FROM enroll WHERE secid = @new_sec;

ถ้าคำสั่งใดเกิด error ให้สั่ง ROLLBACK; แทน COMMIT — ในโปรแกรมจริงจะเขียนใน try/catch หรือใน Stored Procedure ที่มี handler (บทที่ 8)

7.6 การทำงานพร้อมกัน และระดับการแยกรายการ

📝 ข้อสังเกต

SERIALIZABLE ถูกต้องที่สุดแต่ช้าที่สุด — ระบบส่วนใหญ่ใช้ REPEATABLE READ (MySQL) หรือ READ COMMITTED (SQL Server, PostgreSQL)

อาจารย์ A กำลังแก้เกรดนิสิตจาก F เป็น D แต่ยังไม่ COMMIT ระหว่างนั้นเจ้าหน้าที่ B ออกรายงานนิสิตที่สอบตก แล้วไม่พบนิสิตคนนั้น ต่อมาอาจารย์ A กด ROLLBACK — เกิดปัญหาใด และระดับใดต่ำสุดที่ป้องกันได้

Dirty Read — B อ่านค่าที่ยังไม่ถูกยืนยัน ป้องกันได้ตั้งแต่ระดับ READ COMMITTED ขึ้นไป
  • จัดการข้อมูลขั้นสูง (Lab 5.6)
  • ให้นิสิตทุกคนที่เรียนวิชา CS004 ในเทอม 2024-1 ได้เกรด A (UPDATE … WHERE secid IN (SELECT …))
  • เพิ่มเงินเดือน 10% ให้อาจารย์ทุกคนที่สอนในปี 2025 (term LIKE '2025-%')
  • เพิ่มเงินเดือน 20% ให้อาจารย์ที่มีเงินเดือนต่ำกว่าค่าเฉลี่ยเงินเดือนของอาจารย์ทั้งหมด (ระวัง Error 1093) (Derived Table)
