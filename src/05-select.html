5.2 โครงสร้างและลำดับการประมวลผลของ SELECT

รูปแบบเต็ม ([ ] = ละได้)
SELECT   [DISTINCT] รายการคอลัมน์
FROM     ชื่อตาราง
[WHERE    เงื่อนไขการคัดเลือกแถว]
[GROUP BY คอลัมน์ที่ใช้จัดกลุ่ม]
[HAVING   เงื่อนไขการคัดเลือกกลุ่ม]
[ORDER BY คอลัมน์ [ASC | DESC]]
[LIMIT    จำนวนแถว [OFFSET ตำแหน่งเริ่มต้น]];

💡 ทำไมต้องรู้ลำดับ

อธิบายได้ว่าทำไมใช้ชื่อแทน (alias) จาก SELECT ใน WHERE ไม่ได้ (WHERE ทำงานก่อน SELECT) แต่ใช้ใน ORDER BY ได้ (ORDER BY ทำงานหลัง SELECT)

SELECT stdid, gpa * 25 AS score FROM student WHERE score > 90; SELECT stdid, gpa * 25 AS score FROM student ORDER BY score DESC;

ในคำสั่ง SELECT major, COUNT(*) FROM student WHERE gpa > 2 GROUP BY major HAVING COUNT(*) > 1 ORDER BY major; ส่วนใดทำงานเป็นลำดับที่ 2 และ 4

ลำดับที่ 2 = WHERE gpa > 2 (คัดแถว), ลำดับที่ 4 = HAVING COUNT(*) > 1 (คัดกลุ่ม)

5.3 การเลือกคอลัมน์ ชื่อแทน และการกำจัดค่าซ้ำ

SELECT * FROM student;
✅ แนวปฏิบัติที่ดี

* สะดวกตอนทดลอง แต่ไม่ควรใช้ในโปรแกรมจริง — เพิ่มคอลัมน์ใหม่แล้วผลลัพธ์เปลี่ยนโดยโปรแกรมไม่รู้ตัว และดึงข้อมูลเกินจำเป็น

SELECT stdid, name, gpa FROM student; SELECT stdid AS student_id, name AS student_name, gpa AS 'grade point' FROM student; SELECT DISTINCT major FROM student; SELECT lecid, name, salary, salary * 12 AS salary_per_year FROM lecturer;

จงแสดงคู่ (major, credit) ที่ไม่ซ้ำกันจากตาราง subject เรียงตาม major — มีกี่แถว

SELECT DISTINCT major, credit FROM subject ORDER BY major, credit;

DISTINCT ใช้กับทุกคอลัมน์รวมกัน ไม่ใช่เฉพาะคอลัมน์แรก

5.4 การกำหนดเงื่อนไขด้วย WHERE

ตัวดำเนินการความหมายตัวอย่าง
=เท่ากับmajor = 'CS'
<> หรือ !=ไม่เท่ากับmajor <> 'CS'
> < >= <=เปรียบเทียบgpa >= 3.00
AND / OR / NOTตรรกะmajor = 'CS' AND gpa > 3
BETWEEN … AND …อยู่ในช่วง (รวมปลาย)gpa BETWEEN 2 AND 3
IN (…) / NOT IN (…)อยู่ในรายการmajor IN ('CS','Math')
IS NULL / IS NOT NULLเป็นค่าว่างgpa IS NULL
LIKEตรงกับรูปแบบname LIKE 'S%'
SELECT stdid, name, major, gpa FROM student WHERE major = 'CS'; SELECT stdid, name, major, gpa FROM student WHERE gpa > 2.50 AND major = 'English'; SELECT lecid, name, salary FROM lecturer WHERE salary BETWEEN 40000 AND 60000; SELECT subid, name, credit, major FROM subject WHERE major IN ('Math', 'Thai', 'Physics'); SELECT subid, name, major FROM subject WHERE major NOT IN ('CS', 'English');
⚠️ AND มาก่อน OR

WHERE major = 'CS' OR major = 'Math' AND gpa > 3 จะถูกตีความเป็น major = 'CS' OR (major = 'Math' AND gpa > 3) — ใส่วงเล็บให้ชัดเสมอ

จงแสดงรหัส ชื่อ สาขา gpa ของนิสิตสาขา CS หรือ Physics ที่มี gpa ตั้งแต่ 2.50 ขึ้นไป

SELECT stdid, name, major, gpa FROM student WHERE major IN ('CS', 'Physics') AND gpa >= 2.50;

ถ้าเขียน major = 'CS' OR major = 'Physics' AND gpa >= 2.50 โดยไม่ใส่วงเล็บ จะได้นิสิต CS ทุกคน (รวม gpa ต่ำและ NULL) ผิดโจทย์

จงแสดงอาจารย์ที่ไม่ได้อยู่สาขา CS และ Math และมีเงินเดือนมากกว่า 50,000

SELECT lecid, name, major, salary FROM lecturer WHERE major NOT IN ('CS', 'Math') AND salary > 50000;

5.5 การจัดการค่าว่าง และการค้นหาด้วย LIKE

5.5.1 ค่าว่างและการเปรียบเทียบ

NULL คือ “ไม่ทราบค่า” — เปรียบเทียบด้วยตัวดำเนินการปกติจะได้ Unknown เสมอ ต้องใช้ IS NULL / IS NOT NULL

SELECT stdid, name, major, gpa FROM student WHERE gpa IS NULL; SELECT stdid, name, gpa FROM student WHERE gpa = NULL; SELECT stdid, name, major FROM student WHERE major IS NULL;
⚠️ กับดัก NULL

WHERE major <> 'CS' จะไม่รวมนิสิตที่ major เป็น NULL ด้วย — ถ้าต้องการรวม ต้องเขียน WHERE major <> 'CS' OR major IS NULL

SELECT stdid, IFNULL(name, '(ไม่มีชื่อ)') AS name_shown, IFNULL(gpa, 0) AS gpa_shown, COALESCE(major, 'ยังไม่ระบุ') AS major_shown FROM student WHERE gpa IS NULL OR major IS NULL OR name IS NULL;

จงแสดงนิสิตทุกคนที่ไม่ได้อยู่สาขา CS โดยรวมนิสิตที่ยังไม่ระบุสาขาด้วย — ควรได้กี่แถว

SELECT stdid, name, major FROM student WHERE major <> 'CS' OR major IS NULL;

นิสิตทั้งหมด 22 คน อยู่ CS 7 คน จึงควรได้ 15 แถว — ถ้าลืม OR major IS NULL จะได้แค่ 11 แถว

5.5.2 การค้นหาด้วยรูปแบบข้อความ LIKE

รูปแบบความหมายค่าที่ตรง (ในกรณีศึกษา)
'S%'ขึ้นต้นด้วย SSakuragi, Sendoh, Sanji
'%a'ลงท้ายด้วย aRukawa, Nobita, Kira, Killua, Kurapika, Hisoka
'%it%'มี it ตรงไหนก็ได้Mitsui, Nobita
'_o%'ตัวที่ 2 เป็น oNobita, Gon, Zoro
'CS00_'CS00 + อักขระ 1 ตัวCS001–CS009 (ไม่รวม CS010)

% = อักขระกี่ตัวก็ได้ (รวม 0 ตัว), _ = อักขระ 1 ตัวพอดี

SELECT stdid, name, major FROM student WHERE name LIKE 'S%'; SELECT subid, name, credit FROM subject WHERE name LIKE '%Math%';
📝 ผลต่อประสิทธิภาพ

LIKE ที่ขึ้นต้นด้วย % ใช้ดัชนีไม่ได้ ต้องอ่านทั้งตาราง ถ้าตารางใหญ่ควรพิจารณา Full-Text Index

(ก) จงแสดงรายวิชาของสาขาภาษาอังกฤษโดยใช้รหัสวิชา (ขึ้นต้น EN)
(ข) จงแสดงนิสิตที่ชื่อมีตัวอักษรตัวที่สามเป็น k

SELECT subid, name FROM subject WHERE subid LIKE 'EN%'; SELECT stdid, name FROM student WHERE name LIKE '__k%';

MySQL เปรียบเทียบข้อความแบบไม่สนตัวพิมพ์ (collation ลงท้าย _ci) จึงได้ทั้ง k และ K

5.6 การจัดเรียงและจำกัดจำนวนแถว

SELECT stdid, name, gpa FROM student WHERE gpa IS NOT NULL ORDER BY gpa DESC; SELECT stdid, name, major, gpa FROM student WHERE major IS NOT NULL ORDER BY major ASC, gpa DESC;
📝 NULL กับการเรียง

ใน MySQL ค่า NULL ถือว่าน้อยที่สุด — อยู่บนสุดเมื่อ ASC และล่างสุดเมื่อ DESC (สังเกต Sakuragi ในผลข้างบน)

SELECT lecid, name, salary FROM lecturer WHERE salary IS NOT NULL ORDER BY salary DESC LIMIT 5; SELECT lecid, name, salary FROM lecturer WHERE salary IS NOT NULL ORDER BY salary DESC LIMIT 4 OFFSET 3;
⚠️ LIMIT กับค่าที่เท่ากัน

Yoda (t03) และ Yoda (t14) มีเงินเดือน 80,000 เท่ากัน ถ้าจะเอา “อันดับ 1” ด้วย LIMIT 1 จะได้แค่คนเดียวแบบสุ่ม — วิธีที่ถูกคือใช้ Subquery (บทที่ 6)

จงแสดงรายวิชา 3 วิชาที่มีหน่วยกิตมากที่สุด ถ้าหน่วยกิตเท่ากันให้เรียงตามรหัสวิชา

SELECT subid, name, credit FROM subject ORDER BY credit DESC, subid LIMIT 3;

5.7 ฟังก์ชันรวม การจัดกลุ่ม และ HAVING

5.7.1 ฟังก์ชันรวม (Aggregate Function)

ฟังก์ชันหน้าที่กับค่าว่าง
COUNT(*)นับแถวทั้งหมดนับรวม NULL
COUNT(คอลัมน์)นับค่าที่ไม่ใช่ NULLไม่นับ
COUNT(DISTINCT คอลัมน์)นับค่าที่ไม่ซ้ำไม่นับ
SUM / AVGผลรวม / ค่าเฉลี่ยข้าม NULL (ตัวหารของ AVG = จำนวนที่ไม่ NULL)
MAX / MINค่าสูงสุด / ต่ำสุดข้าม NULL
SELECT COUNT(*) AS all_rows, COUNT(gpa) AS has_gpa, COUNT(DISTINCT major) AS n_major FROM student;

นิสิต 22 คน มี gpa แค่ 19 คน และมีสาขาที่ระบุค่าแตกต่างกัน 5 สาขา (ไม่นับ NULL)

SELECT COUNT(*) AS n_lecturer, COUNT(salary) AS has_salary, SUM(salary) AS total_salary, ROUND(AVG(salary), 2) AS avg_salary, MAX(salary) AS max_salary, MIN(salary) AS min_salary FROM lecturer;

จงหาค่าเฉลี่ย gpa สองแบบ
(ก) แบบปกติด้วย AVG
(ข) แบบนับคนที่ไม่มี gpa เป็น 0 — ผลต่างกันเพราะอะไร

SELECT ROUND(AVG(gpa), 4) AS avg_skip_null, ROUND(SUM(IFNULL(gpa, 0)) / COUNT(*), 4) AS avg_null_as_zero FROM student;

AVG ข้าม NULL ตัวหารคือ 19 คน ส่วนแบบ
(ข) ตัวหารคือ 22 คน

5.7.2 การจัดกลุ่มด้วย GROUP BY

SELECT major, COUNT(*) AS n_student, ROUND(AVG(gpa), 2) AS avg_gpa FROM student GROUP BY major ORDER BY major; SELECT term, COUNT(*) AS n_section FROM section GROUP BY term ORDER BY term; SELECT grade, COUNT(*) AS n_enroll FROM enroll GROUP BY grade ORDER BY grade; SELECT lecid, term, COUNT(*) AS n_section FROM section GROUP BY lecid, term ORDER BY lecid, term;
⚠️ กฎสำคัญของ GROUP BY

คอลัมน์ใน SELECT ต้องอยู่ใน GROUP BY หรืออยู่ในฟังก์ชันรวมเท่านั้น — MySQL 8 (โหมด ONLY_FULL_GROUP_BY) จะแจ้ง error

SELECT major, name, COUNT(*) FROM student GROUP BY major;

5.7.3 การคัดเลือกกลุ่มด้วย HAVING

WHERE คัดแถวก่อนจัดกลุ่ม · HAVING คัดกลุ่มหลังคำนวณฟังก์ชันรวม — จึงเป็นที่เดียวที่เขียนเงื่อนไขกับฟังก์ชันรวมได้

SELECT major, COUNT(*) AS n_student FROM student GROUP BY major HAVING COUNT(*) >= 3 ORDER BY n_student DESC; SELECT major, COUNT(*) AS n_student, ROUND(AVG(gpa), 2) AS avg_gpa FROM student WHERE gpa IS NOT NULL GROUP BY major HAVING AVG(gpa) > 2.50 ORDER BY avg_gpa DESC; SELECT secid, COUNT(*) AS n_student FROM enroll GROUP BY secid HAVING COUNT(*) >= 4 ORDER BY n_student DESC, secid;

จงแสดงรหัสนิสิตและจำนวนรายการลงทะเบียน เฉพาะนิสิตที่ลงทะเบียนตั้งแต่ 8 รายการขึ้นไป เรียงจากมากไปน้อย

SELECT stdid, COUNT(*) AS n_enroll FROM enroll GROUP BY stdid HAVING COUNT(*) >= 8 ORDER BY n_enroll DESC, stdid;

จงแสดงรหัสกลุ่มเรียนและจำนวนนิสิตที่ได้เกรด A เฉพาะกลุ่มที่มีคนได้ A ตั้งแต่ 2 คน

SELECT secid, COUNT(*) AS n_A FROM enroll WHERE grade = 'A' GROUP BY secid HAVING COUNT(*) >= 2 ORDER BY secid;

เงื่อนไข grade = 'A' ไม่เกี่ยวกับฟังก์ชันรวม จึงวางใน WHERE (คัดแถวก่อนนับ)

5.8 ฟังก์ชันสเกลาร์ และคำสั่ง CASE

ฟังก์ชันสเกลาร์ทำงานทีละแถว คืนค่า 1 ค่าต่อ 1 แถว

กลุ่มฟังก์ชันหน้าที่
ข้อความUPPER / LOWERตัวพิมพ์ใหญ่ / เล็ก
CONCAT(s1, s2, …)ต่อข้อความ

ตัวเลขROUND(x, n)ปัดเศษ

SELECT stdid, UPPER(name) AS name_upper, CHAR_LENGTH(name) AS name_length, CONCAT(stdid, ' - ', name) AS label FROM student WHERE major = 'CS';
📝 CONCAT กับ NULL

ถ้าค่าใดใน CONCAT เป็น NULL ผลทั้งก้อนเป็น NULL — ใช้ CONCAT(stdid, ' - ', IFNULL(name, '?')) ถ้าต้องการกันไว้

SELECT subid, LEFT(subid, 2) AS major_code, SUBSTRING(subid, 3, 3) AS running_no, name FROM subject; SELECT stdid, name, birthday, YEAR(birthday) AS birth_year, TIMESTAMPDIFF(YEAR, birthday, CURDATE()) AS age FROM student WHERE birthday IS NOT NULL ORDER BY birthday;
📝 ข้อสังเกต

ค่า age เปลี่ยนตามวันที่รันจริง — นี่คือเหตุผลที่เก็บ birthday ไม่เก็บ age (Derived Attribute ในบทที่ 2)

จงแสดงรหัสนิสิต ชื่อ และเดือนเกิด (ตัวเลข) ของนิสิตที่เกิดในปี ค.ศ. 2015 เรียงตามวันเกิด

SELECT stdid, name, MONTH(birthday) AS birth_month FROM student WHERE YEAR(birthday) = 2015 ORDER BY birthday;

5.8.1 คำสั่ง CASE

รูปแบบคำสั่ง
CASE
    WHEN เงื่อนไข1 THEN ค่าที่คืน1
    WHEN เงื่อนไข2 THEN ค่าที่คืน2
    ELSE ค่าปริยาย
END
SELECT stdid, name, gpa, CASE WHEN gpa IS NULL THEN 'ยังไม่มีข้อมูล' WHEN gpa >= 3.50 THEN 'ดีเยี่ยม' WHEN gpa >= 2.50 THEN 'ดี' WHEN gpa >= 2.00 THEN 'พอใช้' ELSE 'ต้องปรับปรุง' END AS level FROM student ORDER BY gpa DESC;

CASE + ฟังก์ชันรวม = ตารางไขว้ (Pivot)

SELECT secid, COUNT(*) AS total, SUM(CASE WHEN grade = 'F' THEN 1 ELSE 0 END) AS n_fail, SUM(CASE WHEN grade <> 'F' THEN 1 ELSE 0 END) AS n_pass FROM enroll WHERE grade IS NOT NULL GROUP BY secid ORDER BY secid;

จงแสดงรหัสวิชา ชื่อวิชา และคอลัมน์ credit_type = “วิชาน้อยหน่วยกิต” เมื่อ < 3, “วิชาปกติ” เมื่อ = 3, “วิชาหนัก” เมื่อ > 3

SELECT subid, name, credit, CASE WHEN credit < 3 THEN 'วิชาน้อยหน่วยกิต' WHEN credit = 3 THEN 'วิชาปกติ' ELSE 'วิชาหนัก' END AS credit_type FROM subject;

จากตาราง student จงนับจำนวนนิสิตในแต่ละสาขาที่มี gpa ≥ 3.00 และ < 3.00 แยกเป็นสองคอลัมน์ (ไม่นับคนที่ gpa เป็น NULL)

SELECT major, SUM(CASE WHEN gpa >= 3.00 THEN 1 ELSE 0 END) AS gpa_3_up, SUM(CASE WHEN gpa < 3.00 THEN 1 ELSE 0 END) AS gpa_below_3 FROM student WHERE gpa IS NOT NULL GROUP BY major ORDER BY major;
