## INNER JOIN (Lab 5.1)
แสดงรหัสและชื่อนิสิตที่เคยเรียนวิชา CS001 โดยไม่ซ้ำ || enroll ⋈ section ⋈ student, DISTINCT
```run max="10"
SELECT DISTINCT s.stdid, s.name
FROM   enroll e
       INNER JOIN section sec ON e.secid = sec.secid
       INNER JOIN student s   ON e.stdid = s.stdid
WHERE  sec.subid = 'CS001';
```
แสดงรหัสและชื่อนิสิตที่เคยเรียนวิชา Programming (ใช้ชื่อวิชาเป็นเงื่อนไข) || JOIN subject
```run max="10"
SELECT DISTINCT s.stdid, s.name
FROM   enroll e
       INNER JOIN section sec ON e.secid  = sec.secid
       INNER JOIN subject sub ON sec.subid = sub.subid
       INNER JOIN student s   ON e.stdid  = s.stdid
WHERE  sub.name = 'Programming';
```

## Self Join และ OUTER JOIN (Lab 5.1–5.2)
แสดงรหัสและชื่อวิชาที่เป็น prerequisite ของวิชา CS002 และของวิชา AI || subject sub JOIN subject pre ON sub.pre = pre.subid
```run
SELECT sub.subid, sub.name, pre.subid AS pre_id, pre.name AS pre_name
FROM   subject sub
       INNER JOIN subject pre ON sub.pre = pre.subid
WHERE  sub.subid = 'CS002' OR sub.name = 'AI';
```
แสดงรหัสและชื่อวิชาทุกวิชาที่มีวิชา Programming เป็น prerequisite || pre.name = 'Programming'
```run
SELECT sub.subid, sub.name
FROM   subject sub
       INNER JOIN subject pre ON sub.pre = pre.subid
WHERE  pre.name = 'Programming';
```
แสดงชื่ออาจารย์และเทอมที่สอนของอาจารย์ทุกคน รวมทั้งอาจารย์ที่ไม่เคยสอน || LEFT JOIN
```run max="10"
SELECT l.name, sec.term
FROM   lecturer l
       LEFT JOIN section sec ON l.lecid = sec.lecid
ORDER  BY l.lecid, sec.term;
```
แสดงชื่ออาจารย์ ชื่อวิชา เทอม ของอาจารย์ทุกคน ถ้าไม่เคยสอนให้แสดง “ไม่เคยสอน” ในคอลัมน์วิชาและเทอม || LEFT JOIN + IFNULL
A: ต้องใช้ LEFT JOIN ทั้งสองครั้ง ถ้า JOIN subject แบบ INNER แถวของอาจารย์ที่ไม่เคยสอนจะหายไป
```run max="10"
SELECT l.name,
       IFNULL(sub.name, 'ไม่เคยสอน') AS subject,
       IFNULL(sec.term, 'ไม่เคยสอน') AS term
FROM   lecturer l
       LEFT JOIN section sec ON l.lecid   = sec.lecid
       LEFT JOIN subject sub ON sec.subid = sub.subid
ORDER  BY sec.term IS NULL DESC, l.lecid;
```
แสดงรหัสนิสิตและรหัสวิชาที่ลงทะเบียนของนิสิตทุกคน รวมทั้งคนที่ไม่เคยลงทะเบียน || LEFT JOIN 2 ครั้ง
```run max="10"
SELECT s.stdid, sec.subid
FROM   student s
       LEFT JOIN enroll  e   ON s.stdid = e.stdid
       LEFT JOIN section sec ON e.secid = sec.secid
ORDER  BY sec.subid IS NULL DESC, s.stdid;
```
แสดงรหัสนิสิต ชื่อวิชา เทอม เกรด ของนิสิตทุกคน รวมทั้งคนที่ไม่เคยลงทะเบียน || LEFT JOIN enroll → section → subject
```run max="10"
SELECT s.stdid, sub.name AS subject, sec.term, e.grade
FROM   student s
       LEFT JOIN enroll  e   ON s.stdid   = e.stdid
       LEFT JOIN section sec ON e.secid   = sec.secid
       LEFT JOIN subject sub ON sec.subid = sub.subid
ORDER  BY sec.term IS NULL DESC, s.stdid;
```
แสดงจำนวนรายวิชาที่นิสิตแต่ละคนลงทะเบียน รวมคนที่ไม่เคยลงให้แสดง 0 || COUNT(e.secid) ไม่ใช่ COUNT(*)
A: <code>COUNT(*)</code> นับแถว NULL ที่ LEFT JOIN เติมให้ด้วย คนที่ไม่เคยลงจะได้ 1 แทน 0 จึงต้องนับคอลัมน์จากฝั่งขวา
```run max="10"
SELECT s.stdid, s.name, COUNT(e.secid) AS n_subject
FROM   student s
       LEFT JOIN enroll e ON s.stdid = e.stdid
GROUP  BY s.stdid, s.name
ORDER  BY n_subject, s.stdid;
```

        FROM   student);

## Subquery (Lab 5.4)
แสดงข้อมูลนิสิตสาขา CS ที่ไม่เคยเรียนวิชา Programming เลย || NOT IN / NOT EXISTS
```run
SELECT *
FROM   student s
WHERE  s.major = 'CS'
  AND  NOT EXISTS
       (SELECT *
        FROM   enroll e
               JOIN section sec ON e.secid   = sec.secid
               JOIN subject sub ON sec.subid = sub.subid
        WHERE  e.stdid = s.stdid
          AND  sub.name = 'Programming');
```

แสดงชื่อ ปีเกิด อายุ ของนิสิตที่อายุน้อยที่สุด และนิสิตที่อายุมากที่สุดใน major CS || = (SELECT MAX(birthday) …)
A: ดูเฉลยข้อสุดท้ายของหมวด Set Operator (คำสั่งที่สอง) — ใช้ Subquery หา MAX/MIN(birthday) ของสาขา CS แล้วรวมด้วย UNION หรือใช้ OR ในคำสั่งเดียว
```run
SELECT name, YEAR(birthday) AS birth_year, TIMESTAMPDIFF(YEAR, birthday, CURDATE()) AS age
FROM   student
WHERE  major = 'CS'
  AND  (birthday = (SELECT MAX(birthday) FROM student WHERE major = 'CS')
    OR  birthday = (SELECT MIN(birthday) FROM student WHERE major = 'CS'));
```

## GROUP BY กับ JOIN (Lab 5.5)
แสดงรหัสนิสิตและจำนวนครั้งที่ลงเรียน เฉพาะนิสิตสาขา CS || JOIN student WHERE major = 'CS' GROUP BY
```run
SELECT e.stdid, COUNT(*) AS n_enroll
FROM   enroll e
       JOIN student s ON e.stdid = s.stdid
WHERE  s.major = 'CS'
GROUP  BY e.stdid;
```
แสดงเกรดและจำนวนครั้งที่ได้เกรดนั้นของนิสิตชื่อ Sakuragi || WHERE s.name = … GROUP BY grade
```run
SELECT e.grade, COUNT(*) AS n
FROM   enroll e
       JOIN student s ON e.stdid = s.stdid
WHERE  s.name = 'Sakuragi'
GROUP  BY e.grade;
```
แสดงข้อมูลทุกคอลัมน์ของอาจารย์ที่เคยสอนมากกว่า 3 ครั้ง || IN (SELECT lecid … HAVING COUNT(*) > 3)
```run
SELECT *
FROM   lecturer
WHERE  lecid IN
       (SELECT lecid
        FROM   section
        GROUP  BY lecid
        HAVING COUNT(*) > 3);
```
แสดงข้อมูลอาจารย์สาขา CS ที่เคยสอนวิชา Programming มากกว่า 2 ครั้ง || HAVING COUNT(*) > 2
```run
SELECT l.lecid, l.name, l.salary, l.major, COUNT(*) AS n_programming
FROM   lecturer l
       JOIN section sec ON l.lecid   = sec.lecid
       JOIN subject sub ON sec.subid = sub.subid
WHERE  l.major = 'CS' AND sub.name = 'Programming'
GROUP  BY l.lecid, l.name, l.salary, l.major
HAVING COUNT(*) > 2;
```
แสดงรหัสอาจารย์ เทอม และจำนวนวิชาที่สอนในเทอมนั้น เฉพาะที่สอนมากกว่า 1 วิชาในเทอมนั้น เรียงตามรหัสอาจารย์ || GROUP BY lecid, term HAVING
```run
SELECT lecid, term, COUNT(DISTINCT subid) AS n_subject
FROM   section
GROUP  BY lecid, term
HAVING COUNT(DISTINCT subid) > 1
ORDER  BY lecid, term;
```
แสดงชื่ออาจารย์ที่สอนกลุ่มเรียนมากที่สุด (ถ้าเท่ากันต้องแสดงทุกคน) || HAVING COUNT(*) = (SELECT MAX(…))
```run
SELECT l.name, COUNT(*) AS n_section
FROM   lecturer l
       JOIN section sec ON l.lecid = sec.lecid
GROUP  BY l.lecid, l.name
HAVING COUNT(*) =
       (SELECT MAX(c)
        FROM   (SELECT COUNT(*) AS c
                FROM   section
                GROUP  BY lecid) AS t);
```
ท้าทาย: แสดง secid ชื่อวิชา ชื่ออาจารย์ จำนวนนิสิต และร้อยละของนิสิตที่ได้ A ในแต่ละกลุ่มเรียน || SUM(CASE WHEN grade = 'A' …) * 100 / COUNT(*)
```run max="10"
SELECT sec.secid, sub.name AS subject, l.name AS lecturer,
       COUNT(*) AS n_student,
       ROUND(SUM(CASE WHEN e.grade = 'A' THEN 1 ELSE 0 END) * 100 / COUNT(*), 1) AS pct_A
FROM   section sec
       JOIN subject  sub ON sec.subid = sub.subid
       JOIN lecturer l   ON sec.lecid = l.lecid
       JOIN enroll   e   ON sec.secid = e.secid
GROUP  BY sec.secid, sub.name, l.name
ORDER  BY sec.secid;
```
ท้าทาย: แสดงนิสิตที่ลงทะเบียนวิชาที่มีวิชาบังคับก่อน แต่ไม่เคยลงวิชาบังคับก่อนนั้นมาก่อน || NOT EXISTS + sub.pre
A: “มาก่อน” เปรียบเทียบด้วยภาคเรียน — ต้องไม่มีการลงวิชา pre ในภาคที่<b>ก่อน</b>ภาคที่ลงวิชานี้ (รูปแบบ term 'YYYY-n' เทียบเป็นข้อความได้)
```run max="10"
SELECT s.stdid, s.name, sub.name AS subject, sec.term, sub.pre
FROM   enroll e
       JOIN student s   ON e.stdid   = s.stdid
       JOIN section sec ON e.secid   = sec.secid
       JOIN subject sub ON sec.subid = sub.subid
WHERE  sub.pre IS NOT NULL
  AND  NOT EXISTS
       (SELECT *
        FROM   enroll e2
               JOIN section sec2 ON e2.secid = sec2.secid
        WHERE  e2.stdid    = e.stdid
          AND  sec2.subid  = sub.pre
          AND  sec2.term   < sec.term)
ORDER  BY s.stdid, sec.term;
```

A: อ้าง lecturer ใน Subquery ตรง ๆ จะเกิด Error 1093 ต้องครอบด้วย Derived Table ให้ MySQL คำนวณค่าเฉลี่ยเก็บไว้ก่อน

สร้างตาราง fail_log แล้วคัดลอกการลงทะเบียนที่ได้ F ทั้งหมด พร้อมชื่อนิสิตและชื่อวิชาเข้าไป || INSERT … SELECT … JOIN
```run show="SELECT * FROM fail_log"
CREATE TABLE fail_log (
    secid         INT,
    stdid         VARCHAR(10),
    student_name  VARCHAR(50),
    subject_name  VARCHAR(30),

       JOIN student s   ON e.stdid   = s.stdid

```
