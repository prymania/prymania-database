DDL ฐานข้อมูลและโครงสร้างตาราง

งานที่ต้องการทำคำสั่ง MySQLบท
สร้างฐานข้อมูลCREATE DATABASE ชื่อ CHARACTER SET utf8mb4;4
เลือกฐานข้อมูลUSE ชื่อ;4
ดูรายชื่อตารางSHOW TABLES;4
ดูโครงสร้างตารางDESCRIBE ชื่อตาราง;4
เพิ่มคอลัมน์ALTER TABLE t ADD COLUMN c ชนิด;4
แก้ชนิดคอลัมน์ALTER TABLE t MODIFY COLUMN c ชนิดใหม่;4
เปลี่ยนชื่อคอลัมน์ALTER TABLE t CHANGE COLUMN เก่า ใหม่ ชนิด;4
ลบคอลัมน์ALTER TABLE t DROP COLUMN c;4
ลบตารางDROP TABLE IF EXISTS t;4
ลบข้อมูลทั้งตาราง (คงโครงสร้าง)TRUNCATE TABLE t;4
โครง CREATE TABLE ครบทุกข้อกำหนด
CREATE TABLE ชื่อตาราง (
    id        INT AUTO_INCREMENT,
    code      VARCHAR(10) NOT NULL,
    amount    DECIMAL(10,2) DEFAULT 0,
    ref_id    VARCHAR(10),
    CONSTRAINT pk_x    PRIMARY KEY (id),
    CONSTRAINT uk_x    UNIQUE (code),
    CONSTRAINT chk_x   CHECK (amount >= 0),
    CONSTRAINT fk_x    FOREIGN KEY (ref_id) REFERENCES ตารางแม่(pk)
                       ON DELETE CASCADE      -- หรือ SET NULL / RESTRICT
);

DML เพิ่ม แก้ ลบข้อมูล

งานที่ต้องการทำคำสั่ง MySQLบท
เพิ่มข้อมูลINSERT INTO t (c1, c2) VALUES (v1, v2), (v3, v4);4
เพิ่มจากผล SELECTINSERT INTO t2 (c1, c2) SELECT a, b FROM t1 WHERE …;6
แก้ไขข้อมูลUPDATE t SET c = v WHERE เงื่อนไข;4
แก้ไขตามเงื่อนไขหลายแบบUPDATE t SET c = CASE WHEN … THEN … ELSE … END;6
ลบข้อมูลDELETE FROM t WHERE เงื่อนไข;4
เลขล่าสุดของ AUTO_INCREMENTSELECT LAST_INSERT_ID();4
⚠️ จำให้ขึ้นใจ

UPDATE / DELETE ต้องมี WHERE และควร SELECT ด้วยเงื่อนไขเดียวกันดูก่อนทุกครั้ง

DQL SELECT

โครง SELECT เต็ม (เขียนตามลำดับนี้)
SELECT   [DISTINCT] คอลัมน์, ฟังก์ชันรวม(...) AS ชื่อแทน
FROM     ตาราง1 t1
         [INNER | LEFT] JOIN ตาราง2 t2 ON t1.fk = t2.pk
WHERE    เงื่อนไขของแถว
GROUP BY คอลัมน์ที่ใช้จัดกลุ่ม
HAVING   เงื่อนไขของกลุ่ม (ใช้ฟังก์ชันรวมได้)
ORDER BY คอลัมน์ [ASC | DESC]
LIMIT    จำนวน [OFFSET ข้าม];

DCL ผู้ใช้ สิทธิ์ Procedure Trigger

งานที่ต้องการทำคำสั่งบท
สร้างผู้ใช้CREATE USER 'u'@'localhost' IDENTIFIED BY 'pw';10
ให้สิทธิ์GRANT SELECT, INSERT ON db.* TO 'u'@'localhost';10
ยกเลิกสิทธิ์REVOKE ALL PRIVILEGES ON db.* FROM 'u'@'localhost';10
เรียกโพรซีเยอร์CALL sp(พารามิเตอร์, @out);11
สำรองข้อมูลmysqldump -u root -p db > backup.sql10
โครง Stored Procedure
DELIMITER $$
CREATE PROCEDURE sp_name (IN p_in VARCHAR(10), OUT p_out INT)
BEGIN
    DECLARE v INT DEFAULT 0;
    SELECT COUNT(*) INTO v FROM ตาราง WHERE คอลัมน์ = p_in;
    IF v > 0 THEN
        SET p_out = v;
    ELSE
        SET p_out = 0;
    END IF;
END $$
DELIMITER ;
โครง Trigger
DELIMITER $$
CREATE TRIGGER tg_name
BEFORE INSERT ON ตาราง          -- BEFORE | AFTER  ×  INSERT | UPDATE | DELETE
FOR EACH ROW
BEGIN
    IF NEW.คอลัมน์ < 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'ข้อความแจ้งเตือน';
    END IF;
END $$
DELIMITER ;
